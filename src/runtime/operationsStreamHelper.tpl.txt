/**
 * Represents a streamed __CLIENT_NAME__ response.
 * @typeParam RecordType Parsed record DTO yielded by the single-pass upstream iterator.
 * @typeParam HeadersType HTTP headers captured before the first record is parsed.
 * @constraints The structural default permits WSDL DTOs named Record without shadowing a utility type.
 * @remarks requestRaw contains the serialized SOAP request when populated.
 */
export type StreamOperationResponse<RecordType, HeadersType = {[key: string]: unknown}> = {
  records: AsyncIterable<RecordType>;
  headers: HeadersType;
  requestRaw?: string;
};
